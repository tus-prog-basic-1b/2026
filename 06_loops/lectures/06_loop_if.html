繰り返し文 × if 文（for / while / 多重ループ）¶

ここでは、 for / while と if 文 の組み合わせ方、および 多重ループ における条件分岐の使い方を学びます。 条件分岐（if）を繰り返しの内側に置くことで、必要なときだけ処理を実行したり、出力や計算を切り替えられます。

if文のおさらい¶

if (条件式) {
    処理A;
} else if (別の条件式) {
    処理B;
} else {
    処理C;
}
  • ==（等しい）、!=（等しくない）、<、<=、>、>= が関係演算子。
  • &&（かつ）、||（または）、!（否定）が論理演算子。

for × if：条件付きの反復¶

1〜100 のうち、 3の倍数のみ を表示：

#include <stdio.h>

int main(void){
    for(int i = 1; i <= 100; i++){
        if(i % 3 == 0){
            printf("%d\n", i);
        }
    }
    return 0;
}

if文の条件を i % 3 == 0 とすることで、i が3で割り切れるとき（3の倍数のとき）だけ printf() が実行されます。

while × if：条件が続く限り判定し続ける¶

正の整数を読み取り、 偶数だけ 合計。0が入力されたら終了：

#include <stdio.h>

int main(void){
    int x, sum = 0;
    printf("正の整数（0で終了）を入力: ");
    scanf("%d", &x);
    while(x != 0){
        if(x > 0 && x % 2 == 0){
            sum += x;
        }
        printf("次: ");
        scanf("%d", &x);
    }
    printf("偶数の合計: %d\n", sum);
    return 0;
}

while文の継続条件式 x != 0 により、0 が入力されるまで読み取りを繰り返します。if文の条件を x > 0 && x % 2 == 0 とすることで、正の偶数が入力されたときだけ sum に足されます。

多重ループ × if：場所に応じて出力を変える¶

例1：枠付き長方形（縦H×横W）¶

外枠を#、内側を空白にする：

#include <stdio.h>

int main(void){
    int H = 5, W = 12;
    for(int i = 0; i < H; i++){
        for(int j = 0; j < W; j++){
            if(i == 0 || i == H-1 || j == 0 || j == W-1){
                printf("#");
            }else{
                printf(" ");
            }
        }
        printf("\n");
    }
    return 0;
}

if文の条件を i == 0 || i == H-1 || j == 0 || j == W-1 とすることで、最初の行・最後の行・最初の列・最後の列のとき（外枠のとき）だけ # が、それ以外のときは空白が出力されます。

例2：斜線と対角線¶

i == j を \、i + j == N-1 を /、それ以外を .：

#include <stdio.h>

int main(void){
    int N = 9; // 正方形
    for(int i = 0; i < N; i++){
        for(int j = 0; j < N; j++){
            if(i == j){
                printf("\\");
            }else if(i + j == N - 1){
                printf("/");
            }else{
                printf(".");
            }
        }
        printf("\n");
    }
    return 0;
}

i == j は左上から右下への対角線上、i + j == N - 1 は右上から左下への対角線上の位置を表します。N=9 の中央 (i, j) = (4, 4) では両方の条件が真になりますが、先にある if が選ばれるため、\ が出力されます。else if は、それより前の条件が偽のときだけ判定されます。なお、printf() で \ を1文字出力するには、文字列の中に \\ と書きます。

例3：チェッカーボード¶

(i + j) % 2 の偶奇で出力を切り替える：

#include <stdio.h>

int main(void){
    int H = 8, W = 16;
    for(int i = 0; i < H; i++){
        for(int j = 0; j < W; j++){
            if( (i + j) % 2 == 0 ){
                printf("*");
            }else{
                printf(" ");
            }
        }
        printf("\n");
    }
    return 0;
}

(i + j) % 2 == 0 は、行番号と列番号の和が偶数の位置を表します。この位置は1マスおきに現れるため、* と空白が交互に並んだ模様になります。

発展：continue / break（if と併用）¶

continue と break は、主に繰り返し文の中で使います。if文と組み合わせると、条件を満たしたときに、残りの処理をスキップして次の周回を始めたり、繰り返しそのものを途中で終えたりできます。

  • continue;：残りの処理を飛ばして次の周回へ。
  • break;：そのループ自体を直ちに抜ける。

例：1〜15のうち3の倍数以外を出力する¶

#include <stdio.h>

int main(void){
    for(int i = 1; i <= 20; i++){
        if(i % 3 == 0){
            continue;   // 3の倍数はスキップ
        }
        if(i > 15){
            break;      // 16以上になったら終了
        }
        printf("%d ", i);
    }
    printf("\n");
    return 0;
}

i が3の倍数のときは continue で表示を飛ばし、i が15より大きくなったら break で繰り返しを終えます。そのため、1〜15 のうち3の倍数でない数だけが表示されます。
