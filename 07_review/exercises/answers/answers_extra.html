第7回：総合演習 追加問題（解答例）¶

問題へ

(13)〜(24) の解答例である。各コードは問題番号に合わせた別のCファイルとして保存する。解答例は一例であり、仕様を満たしていれば別の書き方でもよい。

(13) 大きい方の値¶

comp13.c に保存する。

#include <stdio.h>

int main(void) {
    int a, b;

    printf("2つの整数を入力してください: ");
    scanf("%d %d", &a, &b);
    int max = (a > b) ? a : b;
    printf("大きい方は %d\n", max);
    return 0;
}

(a > b) ? a : b は、a > b が真なら a、偽なら b の値になる。2つの値が等しいときは b の値になるが、どちらも同じ値なので問題ない。

(14) 3つの値の一部を使って10を作れるか¶

comp14.c に保存する。

#include <stdio.h>

int main(void) {
    int a, b, c;

    scanf("%d %d %d", &a, &b, &c);
    if (a == 10 || b == 10 || c == 10 ||
        a + b == 10 || a + c == 10 || b + c == 10 ||
        a + b + c == 10) {
        printf("Yes\n");
    } else {
        printf("No\n");
    }
    return 0;
}

選び方は、1つを選ぶ3通り、2つを選ぶ3通り、3つすべての1通りの、合わせて7通りある。どれか1つでも10になれば Yes なので、7つの条件を || でつなぐ。

(15) 3の5乗¶

comp15.c に保存する。

#include <stdio.h>

int main(void) {
    int p = 1;

    for (int i = 1; i <= 5; i++) {
        p = p * 3;
    }
    printf("%d\n", p);
    return 0;
}

出力例：

243

p を 1 から始めて、3 を5回掛ける。p を 0 から始めると、何を掛けても 0 のままになる。

(16) 漸化式¶

comp16.c に保存する。

#include <stdio.h>

int main(void) {
    int x = 2;

    for (int n = 1; n <= 5; n++) {
        printf("%d\n", x);
        x = x * 3 + 1;
    }
    return 0;
}

出力例：

2
7
22
67
202

変数 x に今の項を入れておき、表示した後で x = x * 3 + 1; として次の項に更新する。

(17) 枠付き長方形¶

comp17.c に保存する。

#include <stdio.h>

int main(void) {
    int H = 4, W = 10;

    for (int i = 0; i < H; i++) {
        for (int j = 0; j < W; j++) {
            if (i == 0 || i == H - 1 || j == 0 || j == W - 1) {
                printf("#");
            } else {
                printf(" ");
            }
        }
        printf("\n");
    }
    return 0;
}

出力例：

##########
#        #
#        #
##########

最初の行（i == 0）・最後の行（i == H - 1）・最初の列（j == 0）・最後の列（j == W - 1）のどれかに当てはまるときに # を、それ以外は空白を出力する。条件は ||（または）でつなぐ。

(18) 複数入力版 FizzBuzz¶

comp18.c に保存する。

#include <stdio.h>

int main(void) {
    int X, Y, N;

    scanf("%d %d %d", &X, &Y, &N);
    for (int i = 1; i <= N; i++) {
        if (i % X == 0 && i % Y == 0) {
            printf("FizzBuzz\n");
        } else if (i % X == 0) {
            printf("Fizz\n");
        } else if (i % Y == 0) {
            printf("Buzz\n");
        } else {
            printf("%d\n", i);
        }
    }
    return 0;
}

X の倍数かつ Y の倍数かどうかを最初に判定する。i % 15 == 0 のように数値を決めて書くと、X、Y を変えたときに正しく動かない。

(19) 5から1までの総和¶

comp19.c に保存する。

#include <stdio.h>

int main(void) {
    int N = 5;
    int s = 0;

    for (int i = N; i >= 1; i--) {
        s = s + i;
    }
    printf("%d\n", s);
    return 0;
}

継続条件式が i > 1 だと、i が 1 のときに繰り返しを終えるので、1 が足されず 14 になる。1 まで含めるには i >= 1 にする。警告は出ないので、期待する結果（15）と比べて誤りに気づく。

(20) N以下の偶数¶

comp20.c に保存する。

#include <stdio.h>

int main(void) {
    int N = 5;
    int i = 2;

    while (i <= N) {
        printf("%d\n", i);
        i += 2;
    }
    return 0;
}

-Wall -Wextra を付けると、'i' is used uninitialized（i に値を入れずに使っている）という警告が出る。正の偶数は 2 から始まるので、int i = 2; とする。int i = 1; とすると、1、3、5 が表示されてしまう。

(21) 0 が入力されるまで繰り返す¶

comp21.c に保存する。

#include <stdio.h>

int main(void) {
    int choice = 0;

    do {
        printf("番号を入力してください（0で終了）: ");
        scanf("%d", &choice);
        printf("%d が選ばれました\n", choice);
    } while (choice != 0);
    printf("終了します\n");
    return 0;
}

choice = 0 は比較ではなく代入なので、choice に 0 が入り、条件はいつも偽になる。そのため、何を入力しても1回で終わってしまう。-Wall -Wextra を付けると、suggest parentheses around assignment used as truth value という警告が出る。「0 でない間は繰り返す」ので、choice != 0 にする。

(22) Good / Bad / Perfect!¶

comp22.c に保存する。

#include <stdio.h>

int main(void) {
    for (int i = 1; i <= 30; i++) {
        if (i % 15 == 0) {
            printf("Perfect! ");
        } else if (i % 3 == 0) {
            printf("Good ");
        } else if (i % 5 == 0) {
            printf("Bad ");
        } else {
            printf("%d ", i);
        }
    }
    printf("\n");
    return 0;
}

3と5の両方の倍数（15の倍数）を最初に判定する。先に3の倍数を判定すると、15 や 30 で「Good」と表示されてしまう。

(23) ひし形¶

comp23.c に保存する。

#include <stdio.h>

int main(void) {
    int N = 3;

    for (int i = 1; i <= N; i++) {
        for (int j = 1; j <= N - i; j++) {
            printf(" ");
        }
        for (int j = 1; j <= 2 * i - 1; j++) {
            printf("*");
        }
        printf("\n");
    }
    for (int i = N - 1; i >= 1; i--) {
        for (int j = 1; j <= N - i; j++) {
            printf(" ");
        }
        for (int j = 1; j <= 2 * i - 1; j++) {
            printf("*");
        }
        printf("\n");
    }
    return 0;
}

出力例：

  *
 ***
*****
 ***
  *

i 行目（上半分）では、空白を N - i 個、* を 2 * i - 1 個並べる。下半分は、同じ並べ方を i = N − 1 から 1 まで減らしながら繰り返す。

(24) 表形式 FizzBuzz¶

comp24.c に保存する。

#include <stdio.h>

int main(void) {
    int R = 4, C = 5;
    int n = 1;

    for (int i = 0; i < R; i++) {
        for (int j = 0; j < C; j++) {
            if (n % 15 == 0) {
                printf("FB ");
            } else if (n % 3 == 0) {
                printf(" F ");
            } else if (n % 5 == 0) {
                printf(" B ");
            } else {
                printf("%2d ", n);
            }
            n++;
        }
        printf("\n");
    }
    return 0;
}

出力例：

 1  2  F  4  B
 F  7  8  F  B
11  F 13 14 FB
16 17  F 19  B

(11) の長方形の並べ方と同じく、表示する値を n で数える。数値を表示する代わりに、FizzBuzz の判定をして F・B・FB を表示する。どれも %2d と同じ2文字分の幅になるように、" F "、"FB " のように空白を付ける。
